    sealed class Spec
    {
        public string View;
        public Dictionary<int, int> Retype = new Dictionary<int, int>();
        public HashSet<int> Main = new HashSet<int>();
        public HashSet<int> Remove = new HashSet<int>();
        // (template index, between index a, between index b, weapon type) — added at the midpoint of a and b.
        public List<(int template, int a, int b, int type)> Add = new List<(int, int, int, int)>();
        public List<(HardPointType type, int template, Vector3 position)> AddSystems = new List<(HardPointType, int, Vector3)>();
        public List<int> RegisterExtra = new List<int>();
        public int ExtraType;
        public System.Action<GameObject, List<HardPoint>> Custom;
    }

    static Dictionary<int, int> Range(int from, int to, int type)
    {
        var map = new Dictionary<int, int>();
        for (int i = from; i <= to; i++) map[i] = type;
        return map;
    }

    static Spec S(string view) => new Spec { View = view };

    static Spec With(this Spec spec, Dictionary<int, int> map)
    {
        foreach (var pair in map) spec.Retype[pair.Key] = pair.Value;
        return spec;
    }

    static Spec Set(this Spec spec, int type, params int[] indices)
    {
        foreach (int i in indices) spec.Retype[i] = type;
        return spec;
    }

    static Spec Drop(this Spec spec, params int[] indices)
    {
        foreach (int i in indices) spec.Remove.Add(i);
        return spec;
    }

    static Spec MainBattery(this Spec spec, params int[] indices)
    {
        foreach (int i in indices) spec.Main.Add(i);
        return spec;
    }

    static Spec Between(this Spec spec, int a, int b, int type)
    {
        spec.Add.Add((a, a, b, type));
        return spec;
    }

    public static string Run(string filter)
    {
        var report = new StringBuilder();
        foreach (Spec spec in Specs())
        {
            if (!string.IsNullOrEmpty(filter) && !spec.View.Contains(filter)) continue;
            report.AppendLine(Apply(spec));
        }
        AssetDatabase.SaveAssets();
        return report.ToString();
    }

    static string Apply(Spec spec)
    {
        GameObject root = PrefabUtility.LoadPrefabContents(spec.View);
        try
        {
            // Editor-only authoring of a prefab hierarchy.
            EntityBindings bindings = root.GetComponentInChildren<EntityBindings>(true);
            var registry = new ShipHardPointRegistry(bindings);
            List<HardPoint> units = registry.HardPoints.ToList();
            var report = new StringBuilder(System.IO.Path.GetFileNameWithoutExtension(spec.View) + ":");

            spec.Custom?.Invoke(root, units);

            foreach (var pair in spec.Retype) SetWeapon(units[pair.Key], pair.Value);
            foreach (int index in spec.Main) SetMain(units[index], true);

            foreach (var add in spec.Add)
            {
                Vector3 position = (units[add.a].transform.position + units[add.b].transform.position) * 0.5f;
                HardPoint template = units[add.template];
                HardPoint created = registry.AddClone(template, template.name + "_" + add.type, position);
                SetWeapon(created, add.type);
                SetMain(created, false);
                if (add.type == PT || add.type == CM || add.type == DPD) SetArc(created, -180f, 180f);
                report.Append($" +{add.type}");
            }

            foreach (var system in spec.AddSystems)
            {
                HardPoint template = units[system.template];
                Vector3 world = root.transform.TransformPoint(system.position);
                HardPoint created = template.HardPointType == system.type
                    ? registry.AddClone(template, template.name + "_2", world)
                    : registry.AddNew(system.type, null, template.transform.parent, system.type.ToString(), world);
                report.Append($" +{system.type}");
            }

            if (spec.RegisterExtra.Count > 0)
            {
                foreach (WeaponHardPoint extra in ShipHardPointValidator.FindUnregistered(registry).OfType<WeaponHardPoint>())
                {
                    registry.Register(extra);
                    SetWeapon(extra, spec.ExtraType);
                    report.Append($" reg {extra.name}");
                }
            }

            foreach (int index in spec.Remove.OrderByDescending(i => i))
            {
                HardPoint point = units[index];
                report.Append($" -{point.name}");
                registry.Remove(point);
            }

            registry.Renumber();
            var issues = ShipHardPointValidator.Validate(registry, bindings,
                ShipHardPointValidator.FindShipData(spec.View));
            if (issues.Count > 0) report.Append(" ISSUES: " + string.Join(" | ", issues));
            PrefabUtility.SaveAsPrefabAsset(root, spec.View);
            return report.ToString();
        }
        finally
        {
            PrefabUtility.UnloadPrefabContents(root);
        }
    }

    static void SetWeapon(HardPoint point, int type)
    {
        var serialized = new SerializedObject(point);
        serialized.FindProperty("<WeaponType>k__BackingField").intValue = type;
        serialized.ApplyModifiedPropertiesWithoutUndo();
    }

    static void SetMain(HardPoint point, bool main)
    {
        var serialized = new SerializedObject(point);
        serialized.FindProperty("mainBattery").boolValue = main;
        serialized.ApplyModifiedPropertiesWithoutUndo();
    }

    static void SetArc(HardPoint point, float min, float max)
    {
        var serialized = new SerializedObject(point);
        serialized.FindProperty("yAxisRange.<Min>k__BackingField").floatValue = min;
        serialized.FindProperty("yAxisRange.<Max>k__BackingField").floatValue = max;
        serialized.ApplyModifiedPropertiesWithoutUndo();
    }
